window.PokeLiteCheats = (() => {
  function app() {
    const el =
      document.querySelector('#app') ||
      document.querySelector('[data-v-app]') ||
      document.body.firstElementChild;

    const vueApp = el?.__vue_app__;

    if (!vueApp) {
      throw new Error('App Vue não encontrado. Abra o jogo completamente antes de rodar o cheat.');
    }

    return vueApp;
  }

  function pinia() {
    const a = app();

    const p =
      a.config?.globalProperties?.$pinia ||
      Object.values(a._context?.provides || {}).find(v => v?._s instanceof Map);

    if (!p || !(p._s instanceof Map)) {
      throw new Error('Pinia não encontrado. Espere o jogo carregar completamente.');
    }

    return p;
  }

  function hasStore(id) {
    return pinia()._s.has(id);
  }

  function store(id) {
    const s = pinia()._s.get(id);
    if (!s) throw new Error(`Store '${id}' não encontrado.`);
    return s;
  }

  function runStore() {
    return store('run');
  }

  function metaStore() {
    return store('metaProgression');
  }

  function pokedexStore() {
    return store('pokedex');
  }

  function userDexStore() {
    return hasStore('userPokedex') ? store('userPokedex') : null;
  }

  function activeRun() {
    const r = runStore().active;
    if (!r) throw new Error('Nenhuma run ativa encontrada. Entre numa run primeiro.');
    return r;
  }

  function getTeamContainer() {
    const r = activeRun();

    const candidates = [
      { parent: r.state, key: 'team' },
      { parent: r, key: 'team' },
      { parent: r.state, key: 'party' },
      { parent: r, key: 'party' },
      { parent: r.state, key: 'members' },
      { parent: r, key: 'members' }
    ];

    for (const c of candidates) {
      if (c.parent && Array.isArray(c.parent[c.key])) return c;
    }

    console.log('Run ativa:', r);
    throw new Error('Não achei o array do time. Use PokeLiteCheats.rawRun() para inspecionar.');
  }

  function teamArray() {
    const c = getTeamContainer();
    return c.parent[c.key];
  }

  function setTeamArray(newTeam) {
    const c = getTeamContainer();
    c.parent[c.key] = newTeam;
    touch();
  }

  function member(slot = 1) {
    const p = teamArray()[Number(slot) - 1];

    if (!p) {
      throw new Error(`Não existe Pokémon no slot ${slot}. Use PokeLiteCheats.team().`);
    }

    return p;
  }

  function getBagContainer() {
    const r = activeRun();

    const candidates = [
      { parent: r.state, key: 'inventory' },
      { parent: r, key: 'inventory' },
      { parent: r.state, key: 'bag' },
      { parent: r, key: 'bag' },
      { parent: r.state, key: 'items' },
      { parent: r, key: 'items' },
      { parent: r.state, key: 'bagItems' },
      { parent: r, key: 'bagItems' }
    ];

    for (const c of candidates) {
      if (c.parent && Array.isArray(c.parent[c.key])) return c;
    }

    console.log('Run ativa:', r);
    throw new Error('Não achei a mochila/bag na run ativa.');
  }

  function bagArray() {
    const c = getBagContainer();
    return c.parent[c.key];
  }

  function safeClone(obj) {
    try {
      return structuredClone(obj);
    } catch {
      return JSON.parse(JSON.stringify(obj));
    }
  }

  function speciesIdOf(p) {
    return p.speciesId ?? p.pokemonId ?? p.dexId ?? p.id;
  }

  function rootIdOf(p) {
    return (
      p.rootSpeciesId ??
      p.evoRootSpeciesId ??
      p.lineRootSpeciesId ??
      p.evolutionRootId ??
      p.evolutionRootSpeciesId ??
      speciesIdOf(p)
    );
  }

    const key = String(stat).toLowerCase().replace(/\s+/g, '');

    const map = {
      hp: 'hp',

      atk: 'atk',
      attack: 'atk',
      ataque: 'atk',

      defesa: 'def',

      spattack: 'special',

      spdef: 'spdef',

      defesaespecial: 'spdef'
    };

    if (!map[key]) {
      throw new Error('Stat inválido. Use: hp, atk, def, speed, special ou spdef.');
    }

    return map[key];
  }

    const key = String(itemId).toLowerCase().trim();

    const map = {

  function touch() {
    const rs = runStore();

    try {
      const c = getTeamContainer();
      c.parent[c.key] = [...c.parent[c.key]];
    } catch {}

    try {
      const c = getBagContainer();
      c.parent[c.key] = [...c.parent[c.key]];
    } catch {}

    try { rs.$patch({}); } catch {}

    return '✅ Alteração aplicada.';
  }

  function maxHpFor(p) {
    const hpBase =
      p.baseStats?.hp ??
      p.stats?.hp ??
      p.hpBase ??
      p.baseHp ??
      100;

    const level = Number(p.level || 100);

    const calc = Math.floor((hpBase * level) / 50) + level + 10;
    return Math.max(Number(p.maxHp || 0), calc, 1);
  }

  function setHpFull(p) {
    p.maxHp = maxHpFor(p);
    p.currentHp = p.maxHp;
    p.hp = p.currentHp;
  }

  function makeShiny(p, value = true) {
    const v = !!value;

    // Campos principais usados em estruturas diferentes do jogo
    p.shiny = v;
    p.isShiny = v;
    p.wasShiny = v;

    // Campos de variante/cosmético que alguns Pokémon criados manualmente precisam
    p.variant = v ? 'shiny' : 'normal';
    p.spriteVariant = v ? 'shiny' : 'normal';
    p.cosmetic = v ? 'shiny' : 'normal';

    // Alguns objetos carregam a flag dentro de form
    if (!p.form || typeof p.form !== 'object') {
      p.form = {};
    }

    p.form.shiny = v;
    p.form.isShiny = v;
    p.form.variant = v ? 'shiny' : 'normal';

    // Alguns Pokémon criados por setTeamManual ficam com dados internos aninhados
    for (const key of ['pokemon', 'species', 'base', 'data', 'template']) {
      if (p[key] && typeof p[key] === 'object') {
        p[key].shiny = v;
        p[key].isShiny = v;
        p[key].wasShiny = v;
        p[key].variant = v ? 'shiny' : 'normal';
        p[key].spriteVariant = v ? 'shiny' : 'normal';
      }
    }

    function fixSpriteUrl(url) {
      if (!url || typeof url !== 'string') return url;

      if (v) {
        if (url.includes('/pokemon/shiny/')) return url;
        return url.replace('/pokemon/', '/pokemon/shiny/');
      }

      return url.replace('/pokemon/shiny/', '/pokemon/');
    }

    p.spriteUrl = fixSpriteUrl(p.spriteUrl);
    p.image = fixSpriteUrl(p.image);
    p.iconUrl = fixSpriteUrl(p.iconUrl);

    if (p.sprites && typeof p.sprites === 'object') {
      for (const k of Object.keys(p.sprites)) {
        if (typeof p.sprites[k] === 'string') {
          p.sprites[k] = fixSpriteUrl(p.sprites[k]);
        }
      }
    }

    return p;
  }

  function setItem(p, itemId) {
    if ('heldItemId' in p) p.heldItemId = itemId;
    else if ('itemId' in p) p.itemId = itemId;
    else p.heldItemId = itemId;
  }

  function setMoveType(p, type) {
    p.moveType = type;
  }

  function setMoveTier(p, tier = 2) {
    p.moveTier = Math.max(0, Math.min(2, Number(tier) || 0));
  }

  function applyBasicMax(p, options = {}) {
    const {
      level = 100,
      shiny = true,
      item = null,
      moveType = null,
      moveTier = 2
    } = options;

    p.level = Number(level) || 100;
    makeShiny(p, shiny);
    setMoveTier(p, moveTier);

    if (moveType) setMoveType(p, moveType);
    if (item) setItem(p, item);

    setHpFull(p);

    return p;
  }

  function applyBuffsToMember(p, amount = 10) {
    const meta = metaStore();
    const rootId = rootIdOf(p);
    const gen = activeRun().gen ?? activeRun().state?.gen ?? 0;

    if (typeof meta.applyTeamStatBuff === 'function') {
      for (const stat of ['hp', 'atk', 'def', 'speed', 'special', 'spdef']) {
        for (let i = 0; i < amount; i++) {
          meta.applyTeamStatBuff(rootId, stat, 999999, gen);
        }
      }
    } else {
      console.warn('applyTeamStatBuff não encontrado. Buffs da Battle Tower podem não aplicar.');
    }
  }

  async function loadDex() {
    const dex = pokedexStore();

    if (typeof dex.load === 'function' && !dex.isLoaded) {
      await dex.load();
    }

    const list =
      dex.pokemon ||
      dex.all ||
      dex.list ||
      [];

    if (!Array.isArray(list) || list.length === 0) {
      throw new Error('Pokédex não carregou ou está vazia.');
    }

    return list;
  }

  function findDexPokemon(list, idOrName) {
    const q = String(idOrName).toLowerCase();

    return list.find(p =>
      String(p.id) === q ||
      String(p.speciesId) === q ||
      String(p.pokemonId) === q ||
      String(p.dexId) === q ||
      String(p.name || '').toLowerCase() === q
    );
  }

  function copyDexIntoMember(template, dexPoke) {
    const p = safeClone(template);

    const id =
      dexPoke.speciesId ??
      dexPoke.pokemonId ??
      dexPoke.dexId ??
      dexPoke.id;

    p.speciesId = id;

    if ('pokemonId' in p) p.pokemonId = id;
    if ('dexId' in p) p.dexId = id;

    if ('id' in p && typeof p.id === 'number' && p.id === speciesIdOf(template)) {
      p.id = id;
    }

    p.name = dexPoke.name ?? p.name;
    p.nickname = null;

    if (dexPoke.types) p.types = safeClone(dexPoke.types);
    if (dexPoke.baseStats) p.baseStats = safeClone(dexPoke.baseStats);
    if (dexPoke.stats) p.stats = safeClone(dexPoke.stats);

    const sprite =
      dexPoke.spriteUrl ??
      dexPoke.sprite ??
      dexPoke.image ??
      dexPoke.iconUrl ??
      `https://raw.githubusercontent.com/PokeAPI/sprites/master/sprites/pokemon/${id}.png`;

    p.spriteUrl = sprite;
    p.image = sprite;
    p.iconUrl = sprite;

    if ('rootSpeciesId' in p) p.rootSpeciesId = id;
    if ('evoRootSpeciesId' in p) p.evoRootSpeciesId = id;
    if ('lineRootSpeciesId' in p) p.lineRootSpeciesId = id;
    if ('evolutionRootId' in p) p.evolutionRootId = id;
    if ('evolutionRootSpeciesId' in p) p.evolutionRootSpeciesId = id;

    return p;
  }

  async function makeMemberFromDex(idOrName, options = {}) {
    const currentTeam = teamArray();
    const template = currentTeam[0];

    if (!template) {
      throw new Error('Precisa ter pelo menos 1 Pokémon no time para clonar a estrutura interna.');
    }

    const dex = await loadDex();
    const dexPoke = findDexPokemon(dex, idOrName);

    if (!dexPoke) {
      throw new Error(`Pokémon não encontrado na Pokédex: ${idOrName}`);
    }

    const p = copyDexIntoMember(template, dexPoke);
    applyBasicMax(p, options);

    return p;
  }

  function markCaught(p) {
    const userDex = userDexStore();
    if (!userDex) return;

    const id = speciesIdOf(p);

    try { userDex.markCaught?.(id); } catch {}
    try { userDex.markShinyCaught?.(id); } catch {}

  function readJson(key, fallback) {
    try {
      const raw = localStorage.getItem(key);
      if (!raw) return safeClone(fallback);
      return JSON.parse(raw);
    } catch {
      return safeClone(fallback);
    }
  }

  function writeJson(key, value) {
    localStorage.setItem(key, JSON.stringify(value));
  }

    try { runStore().$patch({}); } catch {}
    try { metaStore().$patch({}); } catch {}

    try {
      const userDex = userDexStore();
      userDex?.$patch?.({});

      if (hasStore('achievements')) store('achievements').$patch?.({});

      if (typeof window.syncToCloud === 'function') window.syncToCloud();
    } catch {}

    try {
      if (typeof window.checkDexAchievements === 'function') window.checkDexAchievements();
    } catch {}

    return true;

  function normalizeAchievementId(a) {
    if (!a) return null;
    if (typeof a === 'string' || typeof a === 'number') return a;

    return (
      a.id ??
      a.key ??
      a.code ??
      a.name ??
      a.slug ??
      null

  function collectAchievementIds() {
    const ids = new Set();

    function addFrom(value) {
      if (!value) return;

      if (Array.isArray(value)) {
        for (const item of value) {
          const id = normalizeAchievementId(item);
          if (id !== null && id !== undefined && id !== '') ids.add(id);
        }
        return;
      }

      if (typeof value === 'object') {
        for (const [key, item] of Object.entries(value)) {
          const id = normalizeAchievementId(item) ?? key;
          if (id !== null && id !== undefined && id !== '') ids.add(id);
        }
      }
    }

    try {
      if (hasStore('achievements')) {
        const a = store('achievements');

        for (const key of [
          'achievements',
          'allAchievements',
          'achievementList',
          'list',
          'all',
          'items',
          'definitions',
          'availableAchievements'
        ]) {
          addFrom(a[key]);
        }
      }
    } catch {}

    for (const key of ['ACHIEVEMENTS', 'achievements', '__ACHIEVEMENTS__', 'POKELITE_ACHIEVEMENTS']) {
      try { addFrom(window[key]); } catch {}
      try { addFrom(globalThis[key]); } catch {}
    }

    return [...ids];
  }

  function currentAchievementSet() {
    const values = [
      ...readJson('poke_achievements', []),
      ...readJson('achievements', []),
      ...readJson('unlockedAchievements', [])
    ];

    try {
      if (hasStore('achievements')) {
        const a = store('achievements');

        for (const key of ['unlocked', 'unlockedAchievements', 'completed', 'completedAchievements']) {
          if (Array.isArray(a[key])) values.push(...a[key]);
          else if (a[key] && typeof a[key] === 'object') values.push(...Object.keys(a[key]));
        }
      }
    } catch {}

    return new Set(values.filter(v => v !== null && v !== undefined && v !== ''));
  }

  function unlockAchievementInStore(id) {
    try {
      if (!hasStore('achievements')) return;

      const a = store('achievements');

      for (const fn of [
        'unlockAchievement',
        'unlock',
        'completeAchievement',
        'complete',
        'markUnlocked',
        'markCompleted'
      ]) {
        if (typeof a[fn] === 'function') {
          try { a[fn](id); } catch {}
        }
      }

      for (const key of ['unlocked', 'unlockedAchievements', 'completed', 'completedAchievements']) {
        if (Array.isArray(a[key]) && !a[key].includes(id)) a[key].push(id);
        else if (a[key] && typeof a[key] === 'object') a[key][id] = true;
      }
    } catch {}
  }

  function unlockDexId(id, shiny = false) {
    const userDex = userDexStore();

    if (shiny) {

    }
  }

  function unlockDexRange(min, max, options = {}) {
    const {
      normal = true,
      shiny = false
    } = options;

    const dex = readJson('poke_dex', {});
    const shinyDex = readJson('poke_shiny_dex', {});

    let normalAdded = 0;
    let shinyAdded = 0;

    for (let id = min; id <= max; id++) {
      if (normal && dex[id] !== 1) {
        dex[id] = 1;
        normalAdded++;
      }

      if (shiny && shinyDex[id] !== 1) {
        shinyDex[id] = 1;
        shinyAdded++;
      }

      unlockDexId(id, shiny);
    }

    writeJson('poke_dex', dex);
    writeJson('poke_shiny_dex', shinyDex);
    syncProgression();

    return {
      range: `${min}-${max}`,
      normalAdded,
      shinyAdded,
      normalTotal: Object.values(dex).filter(v => v === 1).length,
      shinyTotal: Object.values(shinyDex).filter(v => v === 1).length
    };
  }

      writeJson('unlockedAchievements', unlocked);

      for (const [gen, [min, max]] of Object.entries(GEN_RANGES)) {
        let caught = 0;
        let shiny = 0;

        for (let id = min; id <= max; id++) {
          if (dex[id] === 1) caught++;
          if (shinyDex[id] === 1) shiny++;
        }

        gens[`gen${gen}`] = {
          caught,
          shiny,
          total: max - min + 1
        };
      }

      return {
        achievements: achievements.size,
        pokedexCaught: Object.values(dex).filter(v => v === 1).length,
        shinyCaught: Object.values(shinyDex).filter(v => v === 1).length,
        gens
      };
    },

(() => {
  const C = window.PokeLiteCheats;

  if (!C) {
    throw new Error("PokeLiteCheats não encontrado. Rode o script principal primeiro.");
  }

  function getMeta() {
    if (typeof C.rawMeta === "function") {
      return C.rawMeta();
    }

    throw new Error("rawMeta() não encontrado no PokeLiteCheats.");
  }

  function patchMeta(meta) {
    try {
      meta.$patch?.({
        battlePoints: meta.battlePoints
      });
    } catch {}

    try {
      meta.updatedAt = Date.now();
      meta.lastUpdated = Date.now();
    } catch {}

    try {
      if (typeof window.syncToCloud === "function") {
        window.syncToCloud();
      }
    } catch {}

    return true;
  }

  C.bp = function(amount = null) {
    const meta = getMeta();

    if (amount === null || amount === undefined) {
      return {
        battlePoints: Number(meta.battlePoints || 0)
      };
    }

    return C.setBP(amount);
  };

  C.setBP = function(amount = 999999) {
    const meta = getMeta();
    const value = Math.max(0, Number(amount) || 0);

    meta.battlePoints = value;
    patchMeta(meta);

    return {
      message: "✅ BP definido.",
      battlePoints: meta.battlePoints
    };
  };

  C.addBP = function(amount = 1000) {
    const meta = getMeta();
    const value = Number(amount) || 0;

    meta.battlePoints = Math.max(0, Number(meta.battlePoints || 0) + value);
    patchMeta(meta);

    return {
      message: "✅ BP adicionado.",
      added: value,
      battlePoints: meta.battlePoints
    };
  };

  C.maxBP = function(amount = 999999) {
    return C.setBP(amount);
  };

  C.spendBP = function(amount = 0) {
    const meta = getMeta();
    const value = Math.max(0, Number(amount) || 0);

    meta.battlePoints = Math.max(0, Number(meta.battlePoints || 0) - value);
    patchMeta(meta);

    return {
      message: "✅ BP removido.",
      removed: value,
      battlePoints: meta.battlePoints
    };
  };

  console.log("✅ Patch de BP aplicado.");
  console.log("Comandos novos:");
  console.log("PokeLiteCheats.bp()");
  console.log("PokeLiteCheats.setBP(999999)");
  console.log("PokeLiteCheats.addBP(1000)");
  console.log("PokeLiteCheats.maxBP()");
  console.log("PokeLiteCheats.spendBP(100)");
})();

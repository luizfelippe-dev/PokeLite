  function isGamePresent() {
    return typeof state !== 'undefined' && state && Array.isArray(state.team);
  }
  function game() {
    if (!isGamePresent()) {
      throw new Error('Motor do Pokelike nao encontrado. Abra pokelike.xyz, carregue o jogo e rode o script no contexto "top" do Console. Nao use o script antigo com Vue/Pinia.');
    }
    return state;
  }
  function inRun() {
    const s = game();
    if (!s.team.length || !s.runSeed) {
      throw new Error('Entre numa partida e tenha pelo menos um Pokemon antes de usar comandos do time.');
    }
    return s;
  }
  function slotPokemon(slot = 1) {
    const s = inRun();
    const n = Number(slot);
    if (!Number.isInteger(n) || n < 1 || !s.team[n - 1]) {
      throw new Error('Slot invalido. Use PokeLiteCheats.team() para ver os slots disponiveis.');
    }
    return s.team[n - 1];
  }
  function cleanInt(value, fallback = 0) {
    const n = Number(value);
    return Number.isFinite(n) ? Math.floor(n) : fallback;
  }
  function copy(x) {
    return typeof structuredClone === 'function' ? structuredClone(x) : JSON.parse(JSON.stringify(x));
  }
  function normalizeStat(stat) {
    const q = String(stat).toLowerCase().replace(/\s+/g, '');
    const aliases = {
      vida: 'hp', hp: 'hp', ataque: 'atk', attack: 'atk', atk: 'atk',
      defesa: 'def', defense: 'def', def: 'def', velocidade: 'speed', spe: 'speed', speed: 'speed',
      especial: 'special', spa: 'special', spatk: 'special', special: 'special',
      spattack: 'special', defesaespecial: 'spdef', spd: 'spdef', spdefense: 'spdef', spdef: 'spdef'
    };
    if (!aliases[q]) throw new Error('Stat invalido: hp, atk, def, speed, special, spdef.');
    return aliases[q];
  }
  function root(p) {
    return typeof getEvoLineRoot === 'function' ? getEvoLineRoot(p.speciesId) : p.speciesId;
  }
  function refreshUI() {
    const s = game();
    for (const [name, callback] of [
      ['renderTeamBar', () => renderTeamBar(s.team)],
      ['renderItemBadges', () => renderItemBadges(s.items || [])],
      ['refreshTraitPanel', () => refreshTraitPanel()]
    ]) {
      try {
        if ((name === 'renderTeamBar' && typeof renderTeamBar === 'function') ||
            (name === 'renderItemBadges' && typeof renderItemBadges === 'function') ||
            (name === 'refreshTraitPanel' && typeof refreshTraitPanel === 'function')) callback();
      } catch (e) { console.warn('[PokeLiteCheats] Nao foi possivel atualizar ' + name, e); }
    }
  }
  function persist() {
    const s = game();
    if (s.runSeed && typeof runWritesAllowed === 'function' && !runWritesAllowed()) {
      throw new Error('O jogo bloqueou salvar esta partida (outra aba e dona da run). Nao force o salvamento: use a aba principal.');
    }
    if (s.runSeed && typeof saveRun === 'function') saveRun();
    if (s.isEndlessMode && typeof saveEndlessState === 'function') saveEndlessState();
    refreshUI();
    // O jogo tambem pode sincronizar para a nuvem em outras etapas; nao force upload aqui.
    return true;
  }
  function mutate(callback) {
    inRun();
    if (typeof runWritesAllowed === 'function' && !runWritesAllowed()) {
      throw new Error('Esta partida pertence a outra aba; abra a aba proprietaria.');
    }
    const result = callback();
    persist();
    return result;
  }
  function setHp(p, full = true) {
    if (typeof recomputeMaxHpForBuffs === 'function') {
      recomputeMaxHpForBuffs(p);
    } else if (p.baseStats && p.baseStats.hp) {
      const base = typeof calcHp === 'function'
        ? calcHp(p.baseStats.hp, p.level)
        : Math.floor(p.baseStats.hp * p.level / 50) + p.level + 10;
      const buff = p.statBuffs?.hp || 0;
      p.maxHp = Math.max(1, Math.floor(base * (1 + 0.05 * buff)));
    }
    if (full) p.currentHp = Math.max(1, p.maxHp || 1);
  }
  function setShinyFlag(p, shiny = true) {
    p.isShiny = !!shiny;
    p.isAurora = false;
    if (p.speciesId != null) {
      const src = String(p.spriteUrl || `img/sprites/pokemon/${p.speciesId}.png`);
      if (src.includes('/pokemon/')) {
        const base = src.replace(/\/pokemon\/(?:shiny\/|aurora\/)?/, '/pokemon/');
        p.spriteUrl = shiny ? base.replace('/pokemon/', '/pokemon/shiny/') : base;
      } else if (typeof pokeSpritePath === 'function') {
        p.spriteUrl = pokeSpritePath(p.speciesId, p);
      }
    }
    if (shiny && typeof markShinyDexCaught === 'function') {
      markShinyDexCaught(p.speciesId, p.name, p.types, p.spriteUrl);
    }
  }
  function registerCaught(p) {
    if (typeof markPokedexCaught === 'function') markPokedexCaught(p.speciesId);
    if (p.isShiny && typeof markShinyDexCaught === 'function') {
      markShinyDexCaught(p.speciesId, p.name, p.types, p.spriteUrl);
    }
  }
  function readBuffs() {
    if (typeof loadPersistentBuffs === 'function') return loadPersistentBuffs();
    try { return JSON.parse(localStorage.getItem('poke_stat_buffs') || '{}'); }
    catch { return {}; }
  }
  function writeBuffs(obj) {
    if (typeof savePersistentBuffs === 'function') savePersistentBuffs(obj);
    else localStorage.setItem('poke_stat_buffs', JSON.stringify(obj));
  }
  function applyBuffDirect(p, stat, amount, persistent = null) {
    const k = normalizeStat(stat);
    p.statBuffs ||= {};
    const old = Math.max(0, cleanInt(p.statBuffs[k] || 0));
    const next = Math.min(10, old + Math.max(0, cleanInt(amount, 1)));
    p.statBuffs[k] = next;
    if (k === 'atk' || k === 'special') {
      p.statBuffs.atk = Math.max(p.statBuffs.atk || 0, next);
      p.statBuffs.special = Math.max(p.statBuffs.special || 0, next);
    }
    if (k === 'hp') setHp(p, false);
    const buffs = persistent || readBuffs();
    buffs[root(p)] = { ...p.statBuffs };
    if (!persistent) writeBuffs(buffs);
    return { stat: k, previous: old, value: next, max: 10 };
  }
  function allBuffsDirect(p, persistent = null) {
    const db = persistent || readBuffs();
    p.statBuffs ||= {};
    for (const k of STATS) p.statBuffs[k] = 10;
    db[root(p)] = { ...p.statBuffs };
    if (!persistent) writeBuffs(db);
    setHp(p, true);
    return { ...p.statBuffs };
  }
  function itemDefinition(itemId) {
    const id = String(itemId || '').trim().toLowerCase();
    if (HEALING_REPLACEMENTS.has(id)) {
      throw new Error(`O item '${id}' nao existe na mochila da versao 3.7.1. Use reviveAndHealTeam() para curar sem item.`);
    }
    const bag = Array.isArray(game().items) ? game().items : [];
    const pools = [
      ...(typeof ITEM_POOL !== 'undefined' && Array.isArray(ITEM_POOL) ? ITEM_POOL : []),
      ...(typeof USABLE_ITEM_POOL !== 'undefined' && Array.isArray(USABLE_ITEM_POOL) ? USABLE_ITEM_POOL : []),
      ...bag,
      ...game().team.map(p => p?.heldItem).filter(Boolean)
    ];
    return pools.find(x => x?.id?.toLowerCase() === id) || null;
  }
  function readMoney() {
    if (typeof getPokedollars === 'function') return getPokedollars();
    return Math.max(0, cleanInt(localStorage.getItem('poke_pokedollars')));
  }
  function writeMoney(value) {
    const n = Math.max(0, cleanInt(value));
    if (typeof setPokedollars === 'function') setPokedollars(n);
    else localStorage.setItem('poke_pokedollars', String(n));
    return { pokedollars: readMoney(), currency: 'Pokédollars (substitui o antigo BP)' };
  }
  async function findSpecies(input) {
    if (typeof loadStaticPokedex !== 'function' || typeof fetchPokemonById !== 'function' || typeof createInstance !== 'function') {
      throw new Error('Funcoes da Pokédex nao encontradas. Essa versao do jogo pode ter mudado de novo.');
    }
    const entries = await loadStaticPokedex();
    const requested = String(input).trim().toLowerCase();
    let id = Number.isInteger(Number(requested)) && Number(requested) > 0 ? Number(requested) : null;
    if (!id && entries && typeof entries === 'object') {
      const found = Object.entries(entries).find(([key, v]) => String(v?.name || '').toLowerCase() === requested || key.toLowerCase() === requested);
      if (found) id = Number(found[0]);
    }
    const data = await fetchPokemonById(id || requested);
    if (!data?.baseStats || !data?.types) throw new Error('Pokemon nao encontrado na Pokédex: ' + input);
    return data;
  }
  async function buildPokemon(input, options = {}, buffsDb = null) {
    const data = await findSpecies(input);
    const lvl = Math.max(1, cleanInt(options.level ?? 100, 100));
    const shiny = options.shiny !== false;
    const tier = Math.max(0, Math.min(2, cleanInt(options.moveTier ?? 2, 2)));
    const p = createInstance(data, lvl, shiny, tier);
    if (options.item) {
      const item = itemDefinition(options.item);
      if (!item) throw new Error('Item nao encontrado no jogo: ' + options.item);
      p.heldItem = copy(item);
    }
    if (options.moveType) console.warn('[PokeLiteCheats] moveType ignorado: o motor atual calcula o tipo do golpe dinamicamente.');
    if (options.buffs !== 0) {
      const amount = Math.max(0, Math.min(10, cleanInt(options.buffs ?? 10, 10)));
      p.statBuffs = Object.fromEntries(STATS.map(k => [k, amount]));
      const buffMap = buffsDb || readBuffs();
      buffMap[root(p)] = { ...p.statBuffs };
      if (!buffsDb) writeBuffs(buffMap);
      setHp(p, true);
    }
    return p;
  }
  function storageSnapshot() {
    const keys = ['poke_endless_state', 'poke_stat_buffs', 'poke_pokedollars',
      'poke_dex', 'poke_shiny_dex', 'poke_achievements'];
    const values = {};
    for (const k of keys) values[k] = localStorage.getItem(k);
    values.poke_current_run = typeof runSlotGet === 'function'
      ? runSlotGet('poke_current_run') : localStorage.getItem('poke_current_run');
    return values;
  }
  function savedBackup(key) {
    const raw = localStorage.getItem(key);
    if (!raw) throw new Error('Backup nao encontrado: ' + key);
    const data = JSON.parse(raw);
    if (!data || data.source !== 'PokeLiteCheats' || !data.values) throw new Error('Arquivo de backup invalido.');
    return data;
  }

      inRun();
      const index = cleanInt(slot) - 1;
      if (index < 0 || index > 5) throw new Error('Slot deve estar entre 1 e 6.');
      const db = readBuffs();

      const s = game();

      s.maxTeamSize = Math.max(s.maxTeamSize || 1, s.team.length);
      writeBuffs(db); registerCaught(p); persist();

    async setTeamManual(configs = []) {
      inRun();
      if (!Array.isArray(configs) || configs.length === 0 || configs.length > 6) throw new Error('Informe de 1 a 6 Pokemon no array.');
      const db = readBuffs();
      const newTeam = [];
      for (const cfg of configs) newTeam.push(await buildPokemon(cfg.id ?? cfg.name, cfg, db));
      if (typeof runWritesAllowed === 'function' && !runWritesAllowed()) throw new Error('Salvamento bloqueado nesta aba.');
      const s = game();
      s.team = newTeam;
      s.maxTeamSize = Math.max(s.maxTeamSize || 1, newTeam.length);
      writeBuffs(db);
      for (const p of newTeam) registerCaught(p);
      persist();

    unlockAchievements() {
      if (typeof ACHIEVEMENTS === 'undefined' || !Array.isArray(ACHIEVEMENTS)) {
        throw new Error('Lista de conquistas nao encontrada.');
      }
      const before = typeof getUnlockedAchievements === 'function'
        ? getUnlockedAchievements() : new Set(JSON.parse(localStorage.getItem('poke_achievements') || '[]'));
      const ids = ACHIEVEMENTS.map(x => x?.id).filter(Boolean);
      const updated = new Set([...before, ...ids]);
      localStorage.setItem('poke_achievements', JSON.stringify([...updated]));
      return { added: updated.size - before.size, total: updated.size };
    },
    unlockDexGen(gen, options = {}) {
      const range = GEN_RANGES[Number(gen)];
      if (!range) throw new Error('Geracao disponivel no Pokelike 3.7.1: 1 a 6.');
      const norm = options.normal !== false;
      const shiny = !!options.shiny;
      const dex = JSON.parse(localStorage.getItem('poke_dex') || '{}');
      const sh = JSON.parse(localStorage.getItem('poke_shiny_dex') || '{}');
      let addedNormal = 0, addedShiny = 0;
      for (let id = range[0]; id <= range[1]; id++) {
        if (norm && dex[id] !== 1) { dex[id] = 1; addedNormal++; }
        if (shiny && sh[id] !== 1) { sh[id] = 1; addedShiny++; }
      }
      if (norm) localStorage.setItem('poke_dex', JSON.stringify(dex));
      if (shiny) localStorage.setItem('poke_shiny_dex', JSON.stringify(sh));
      if (typeof checkDexAchievements === 'function') { try { checkDexAchievements(); } catch (e) { console.warn(e); } }
      return { gen:Number(gen), range, addedNormal, addedShiny };
    },

    dexStatus() {
      const dex = JSON.parse(localStorage.getItem('poke_dex') || '{}');
      const sh = JSON.parse(localStorage.getItem('poke_shiny_dex') || '{}');
      const achievements = JSON.parse(localStorage.getItem('poke_achievements') || '[]');
      const gens = {};
      for (const [g, [start, end]] of Object.entries(GEN_RANGES)) {
        let caught = 0, shiny = 0;
        for (let n = start; n <= end; n++) { if (dex[n] === 1) caught++; if (sh[n] === 1) shiny++; }
        gens[g] = { caught, shiny, total: end - start + 1 };
      }
      return { gens, totalNormal: Object.values(gens).reduce((a,x) => a+x.caught,0),
        totalShiny: Object.values(gens).reduce((a,x) => a+x.shiny,0), achievements: achievements.length };
    }
